# Lute 0.40.0 — R3 implementation plan

Spec: `docs/proposals/scenario-dsl/0.40.0.md`. Brief: `docs/design/runtime-player.md`
(owner answers 2026-10-11). Work on `feat/0.40`; every slice in its own worktree
under `/tmp/wt-<slice>` with its own `CARGO_TARGET_DIR=/tmp/wt-<slice>-target`;
the orchestrator merges. Gates for every Rust slice: nextest, `-D warnings`,
clippy (pinned 1.96.1), byte identity vs v0.40.0 (P0 harness), bench A/B.

## Slices and dependencies

```text
P0 baseline ─┬─ P1 language+IR ─┬─ P2 runtime+contract ─┬─ P4 player core ─┬─ P6 demo+e2e ─ P7 CI/publish/docs ─ release
             └─ P3 bindings ────┘                        └─ P5 player-web ──┘
```

P4 and P5 start with P1–P3 in flight: they code against the interfaces below
and the spec's wire shapes, then switch to the regenerated contract when P2
lands. P3 provides the generated bindings schema to P4.

## Core ↔ web interface contract (P4 owns the definitions in `packages/player/src/ports/`)

```ts
import type { Effect, Stream, Scope } from "effect"
import type * as Contract from "@lute-lang/runtime/Contract"   // generated
import type { Bindings } from "./bindings"                      // generated from schema

/** A runtime stage (spec §4.1), from Program.stage / SessionState.stage. */
export type Stage = Contract.Stage

/** One decoded record the presenter renders. `timing` is the record's. */
export type CueData =
  | { readonly kind: "line"; readonly line: LineView; readonly timing?: Timing }
  | { readonly kind: "bg" | "music" | "sfx" | "vfx" | "actor" | "camera" | "cg" | "video" | "sequence"
      readonly record: Contract.StagingRecord; readonly timing?: Timing }

export interface LineView {
  readonly lineId: string; readonly voiceKey: string; readonly role: string
  readonly speaker: string; readonly speakerName: string
  readonly segments: ReadonlyArray<Segment>             // resolved for the current locale
  readonly read: boolean                                 // in read history
}
export type Segment =
  | { readonly text: string; readonly styles: ReadonlyArray<string>; readonly rate?: number }
  | { readonly pause: number }
export interface Timing { readonly wait?: boolean; readonly duration?: number; readonly delay?: number; readonly at?: number; readonly timeline?: number }

/** How a cue is played. `skip` = jump to the end state at once. */
export type PlayMode = "normal" | "skip"

export interface Presenter {
  /** Play a cue; completes when its visual effect finished (immediately in skip mode). */
  readonly play: (cue: CueData, mode: PlayMode) => Effect.Effect<void>
  /** Show a line and type it; completes when fully shown. `reveal` completes typing at once. */
  readonly showLine: (line: LineView, mode: PlayMode) => Effect.Effect<void>
  readonly revealLine: Effect.Effect<void>
  readonly clearLine: Effect.Effect<void>
  /** Rebuild the stage at once (load, rollback, new presentation start). */
  readonly restore: (stage: Stage) => Effect.Effect<void>
  /** Choice UI; resolves with the chosen option. Countdown is shown, not decided, here. */
  readonly showChoice: (menu: Contract.MenuAwait, remaining: Stream.Stream<number>) => Effect.Effect<string>
  readonly hideChoice: Effect.Effect<void>
  /** Overlays/screens are driven by the machine state. */
  readonly showScreen: (screen: ScreenModel) => Effect.Effect<void>
  readonly capture: Effect.Effect<string>               // thumbnail PNG data URL
}

export interface AudioPort {
  readonly play: (channel: "music" | "sfx" | "voice", key: AudioKey, options: AudioOptions) => Effect.Effect<void> // completes at sound/fade end
  readonly stop: (channel: "music" | "sfx" | "voice", fade?: number) => Effect.Effect<void>
  readonly setVolumes: (volumes: Volumes) => Effect.Effect<void>
}

export interface AssetLoader {
  readonly ensure: (keys: ReadonlyArray<AssetKey>) => Effect.Effect<void, AssetMissing, Scope.Scope>
  readonly release: (keys: ReadonlyArray<AssetKey>) => Effect.Effect<void>
  readonly progress: Stream.Stream<number>
}

/** Device-independent input. */
export type PlayerAction =
  | { readonly type: "advance" } | { readonly type: "toggleAuto" } | { readonly type: "toggleSkip" }
  | { readonly type: "rollback" } | { readonly type: "openBacklog" } | { readonly type: "openMenu" }
  | { readonly type: "quickSave" } | { readonly type: "quickLoad" } | { readonly type: "hideUI" }
  | { readonly type: "choose"; readonly option: string } | { readonly type: "close" }
  | { readonly type: "screen"; readonly action: string; readonly payload?: unknown }

export interface PlayerHost {
  readonly bridge?: Record<string, (request: Contract.BridgeAwait) => Effect.Effect<Record<string, unknown>>>
  readonly idle?: (state: PlayerView) => Effect.Effect<Contract.Input | undefined>
}
```

`ScreenModel`, `PlayerView` (what screens render: settings, slots, backlog
entries), `AudioKey`, `AssetKey`, `Volumes` are defined in P4 beside these and
shared with P5 by import. P5 MUST NOT import P4 internals beyond `ports/` and
the facade.
